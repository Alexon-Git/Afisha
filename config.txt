server {
    server_name afisha.lalexonl.ru www.afisha.lalexonl.ru;

	root /var/www/alexon/Afisha/frontend/build/;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }
    
    location /static/uploads {
        # files uploaded by backend
        alias /var/www/alexon/Afisha/backend/app/static/uploads/;
        access_log off;
        expires 30d;
    }

    # Serve static files
    location /static/ {
        alias /var/www/alexon/Afisha/frontend/build/static/;
    }

    # Proxy API requests to backend
    location /api/ {
        proxy_pass http://0.0.0.0:8039/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

		rewrite ^/api/v1/([^/]+)$ /api/v1/$1/ break;

	}

   location /events/ {
       proxy_pass http://0.0.0.0:8039/events;
       proxy_set_header Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
   }
    
   location /auth {
       proxy_pass http://0.0.0.0:8039/auth;
       proxy_set_header Host $host;
       proxy_set_header X-Real-IP $remote_addr;
       proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
       proxy_set_header X-Forwarded-Proto $scheme;
   }

    location /health {
        proxy_pass http://0.0.0.0:8039/health;
    }

    error_log /var/log/nginx/afisha-error.log;
    access_log /var/log/nginx/afisha-access.log;

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/afisha.lalexonl.ru/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/afisha.lalexonl.ru/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}



server {
    if ($host = afisha.lalexonl.ru) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name afisha.lalexonl.ru www.afisha.lalexonl.ru;
    return 404; # managed by Certbot


}
